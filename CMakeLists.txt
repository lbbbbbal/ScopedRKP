cmake_minimum_required(VERSION 3.22)
project(ScopedRKP VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Use the Android NDK toolchain with arm64-v8a")
endif()
add_compile_options(-Oz -flto -fPIC -fvisibility=hidden -fstack-protector-strong
  -mbranch-protection=standard -ffunction-sections -fdata-sections)
add_link_options(-flto -Wl,-z,relro,-z,now,-z,noexecstack,-z,max-page-size=16384,--gc-sections,--exclude-libs,ALL,-s)
include_directories(src src/injector)
add_library(scoped_rkp SHARED src/payload.cpp)
target_link_libraries(scoped_rkp dl log)
add_executable(srkp_engine src/injector/main.cpp src/injector/utils.cpp)
target_link_libraries(srkp_engine dl)
